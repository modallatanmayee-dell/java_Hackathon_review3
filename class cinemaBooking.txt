
class CinemaBooking {
    String movieName;
    String customerName;
    double ticketPrice;
    int numberOfTickets;

   
    CinemaBooking() {
        movieName = "AVATAR";
        ticketPrice = 200.00;
        customerName = "Unknown";
        numberOfTickets = 0;
    }

  
    CinemaBooking(String customerName, String movieName,
                  double ticketPrice, int numberOfTickets) {
        this.customerName = customerName;
        this.movieName = movieName;
        this.ticketPrice = ticketPrice;
        this.numberOfTickets = numberOfTickets;
    }

   
    void bookTickets(int tickets) {
        numberOfTickets = tickets;
    }

    
    void bookTickets(String name, int tickets) {
        customerName = name;
        numberOfTickets = tickets;
    }

   
    double calculateTotal() {
        return ticketPrice * numberOfTickets;
    }

   
    double calculateDiscount() {
        if (numberOfTickets >= 5) {
            return calculateTotal() * 0.10;
        }
        return 0.0;
    }

    
    void displayBooking() {
        System.out.println("\nCustomer Name   : "
                + customerName.toUpperCase());
        System.out.println("Movie Name      : "
                + movieName.toUpperCase());

        if (numberOfTickets <= 0) {
            System.out.println("Booking Status  : Invalid number of tickets");
            return;
        }

        System.out.printf("Ticket Price    : Rs. %.2f%n", ticketPrice);
        System.out.println("Number of Tickets: " + numberOfTickets);
        System.out.println("Booking Status  : Booking Successful");

        double total = calculateTotal();
        double discount = calculateDiscount();
        double finalAmount = total - discount;

        System.out.printf("Total Amount    : Rs. %.2f%n", total);
        System.out.printf("Discount        : Rs. %.2f%n", discount);
        System.out.printf("Final Amount    : Rs. %.2f%n", finalAmount);
    }

    // Compare customer names
    boolean compareCustomer(CinemaBooking other) {
        return this.customerName.equalsIgnoreCase(other.customerName);
    }
}

public class Main {
    public static void main(String[] args) {
        
        CinemaBooking b1 = new CinemaBooking(
                "Sneha", "Avatar", 200.00, 5);

        System.out.println("========== CINEMA TICKET BOOKING ==========");
        b1.displayBooking();

        
        CinemaBooking b2 = new CinemaBooking();
        b2.bookTickets("Arjun", 2);

        System.out.println("\n========== SECOND BOOKING ==========");
        System.out.println("Customer Name   : "
                + b2.customerName.toUpperCase());
        System.out.println("Movie Name      : "
                + b2.movieName.toUpperCase());
        System.out.println("Number of Tickets: "
                + b2.numberOfTickets);

       
        System.out.println("\nSame Customer   : "
                + b1.compareCustomer(b2));
    }
}
